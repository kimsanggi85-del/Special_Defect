-- 마스터 데이터 테이블 (발생부서/고객사/품번/발생공정) CREATE TABLE IF NOT EXISTS special_defect_master ( id BIGSERIAL PRIMARY KEY, category TEXT NOT NULL, -- dept, customer, partNo, process item_value TEXT NOT NULL, sort_order INT DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW(), UNIQUE(category, item_value) ); -- RLS 활성화 (사내망 사용 시 정책 별도 설정) ALTER TABLE special_defect_reports ENABLE ROW LEVEL SECURITY; ALTER TABLE special_defect_master ENABLE ROW LEVEL SECURITY; -- 모든 사용자 읽기/쓰기 허용 정책 (사내용) CREATE POLICY "Enable all access for reports" ON special_defect_reports FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "Enable all access for master" ON special_defect_master FOR ALL USING (true) WITH CHECK (true); -- ===== [전자결재] (v2.5 추가) ===== ALTER TABLE special_defect_reports ADD COLUMN IF NOT EXISTS approval_status TEXT DEFAULT '작성중', ADD COLUMN IF NOT EXISTS approval_step SMALLINT DEFAULT 0, ADD COLUMN IF NOT EXISTS approval_line JSONB; UPDATE special_defect_reports SET approval_status = '작성중' WHERE approval_status IS NULL; UPDATE special_defect_reports SET approval_step = 0 WHERE approval_step IS NULL; CREATE TABLE IF NOT EXISTS special_defect_approval_log ( id BIGSERIAL PRIMARY KEY, report_id BIGINT REFERENCES special_defect_reports(id) ON DELETE CASCADE, action TEXT NOT NULL, -- 상신 / 승인 / 반려 / 회수 step_no SMALLINT, step_role TEXT, -- 담당 / 검토 / 승인 actor_name TEXT, actor_dept TEXT, comment TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE INDEX IF NOT EXISTS idx_sd_approval_log_report ON special_defect_approval_log(report_id); ALTER TABLE special_defect_approval_log ENABLE ROW LEVEL SECURITY; CREATE POLICY "Enable all access for approval log" ON special_defect_approval_log FOR ALL USING (true) WITH CHECK (true);
